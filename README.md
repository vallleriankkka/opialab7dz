# Домашнее задание к работе 7
## Условия задания
6. Дано целое число k, определить каким днем недели является k-й день невисокосного года, в котором 1 января приходиться на заданный заранее день недели
### Контрольный пример
```
1 января - 4 (Среда)
Номер дня - 4
Вывод: Понедельник
```
## 1. Алгоритм и блок-схема
### Алгоритм:
1. Начало
2. "Введите день недели на 1 января и номер дня k"
3. Ввод jan, k
4. day = (jan + k - 1) % 7
5. day = 0?\
   Да: "Понедельник"\
   Нет: day = 1?\
   Да: "Вторник"\
   Нет: day = 2?\
   Да: "Среда"\
   Нет: day = 3?\
   Да: "Четверг"\
   Нет: day = 4?\
   Да: "Пятница"\
   Нет: day = 5?\
   Да: "Суббота"\
   Нет: day = 6?\
   Да: "Воскресенье"\

## 2. Реализация программы
```c
#include <locale.h>
#include <stdio.h>

int main()
{
	setlocale(LC_CTYPE, "");
	dz();
}
int dz()
{
	//Контрольный пример
	//1 января - 4 (Среда)
	//Номер дня - 4
	//Вывод: Понедельник
	int k, day, jan;
	puts("Номер дня недели на 1 января и номер дня k");
	scanf("%d %d", &jan, &k);
	day = (jan + k - 1) % 7;
	switch (day)
	{
	case 0:
		puts("Понедельник"); 
		break;
	case 1: 
		puts("Вторник");  
		break;
	case 2: 
		puts("Среда");   
		break;
	case 3: 
		puts("Четверг");
		break;
	case 4: 
		puts("Пятница");
		break;
	case 5: 
		puts("Суббота"); 
		break;
	case 6: 
		puts("Воскресенье");
		break;
	default: 
		puts("?");       
		break;
	}
	return 0;
}
```
